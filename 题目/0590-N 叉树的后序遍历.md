void pS(Node* root,vector<int>& res) {  
    if (root == nullptr) return;  
    for (int i = 0; i < root->children.size(); i++) {  
        pO(root->children[i], res);  
    }  
    res.push_back(root->val);  
}  
vector<int> postorder(Node* root) {  
    vector<int> res;  
    pS(root, res);  
    return res;  
}
