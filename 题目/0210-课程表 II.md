# 210. 课程表 II

[LeetCode 链接](https://leetcode.cn/problems/course-schedule-ii/)

## 来源：灵茶山艾府和代码随想录 / 图

```python
class Solution:
    def findOrder(self, numCourses: int, prerequisites: List[List[int]]) -> List[int]:
        graph = defaultdict(list)
        degree = [0] * numCourses
        queue = deque()
        result = []
        for course, pre in prerequisites:
            graph[pre].append(course)
            degree[course] += 1
        for index, value in enumerate(degree):
            if value == 0:
                queue.append(index)
        while len(queue) != 0:
            index = queue.popleft()
            for value in graph[index]:
                degree[value] -= 1
                if degree[value] == 0:
                    queue.append(value)
            result.append(index)
        if len(result) == numCourses:
            return result
        else:
            return []

```
