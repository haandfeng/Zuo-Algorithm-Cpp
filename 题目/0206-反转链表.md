三个变量反转链表
```c++
class Solution {  
public:  
    ListNode* reverseList(ListNode* head) {  
        if (head == nullptr) return nullptr;  
        if (head->next == nullptr) return head;  
        ListNode* cur = head, *prev = nullptr, *next = nullptr;  
        while (cur) {  
            next = cur->next;  
            cur->next = prev;  
            prev = cur;  
            cur = next;  
        }  
        return prev;  
    }  
};
```
