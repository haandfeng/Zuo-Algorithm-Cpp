# 52. N 皇后 II

[LeetCode 链接](https://leetcode.cn/problems/n-queens-ii/)

## 来源：灵茶山艾府和代码随想录 / 回溯

和上面的题差不多，这里采用0x3f的思路
```c++
class Solution {  
public:  
    int res = 0;  
    bool isValid(int* chessBoard, int row, int col,int n) {  
        for (int i = 0; i < row; i++) {  
            if (chessBoard[i] == col) {  
                return false;  
            }  
            if (i-chessBoard[i] == row -col ) {  
                return false;  
            }  
            if (i+chessBoard[i] == col+row) {  
                return false;  
            }  
        }  
  
        return true;  
    }  
    void backtracking(int* chessBoard,int row,int n) {  
        if (row == n) {  
            res++;  
            return;  
        }  
        for (int i = 0; i < n; i++) {  
            if (isValid(chessBoard, row, i,n)) {  
                chessBoard[row] = i;  
                backtracking(chessBoard, row + 1, n);  
            }  
        }  
    }  
    int totalNQueens(int n) {  
        int * chessBoard = new int[n];  
        backtracking(chessBoard, 0, n);  
        delete[] chessBoard;  
        return res;  
    }  
};
```
