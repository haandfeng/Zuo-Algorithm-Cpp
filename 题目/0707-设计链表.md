# 707. 设计链表

[LeetCode 链接](https://leetcode.cn/problems/design-linked-list/)

## 来源：labuladong / 双指针-链表 和 链表代码实现

主要是用tail head进行优化， 要判断list里面有没有元素，这样才可以处理到一些错误。
```go
type MyLinkedList struct {
	head *Node
	tail *Node
	size int
}

type Node struct {
	val  int
	next *Node
	prev *Node
}

func Constructor() MyLinkedList {
	head := &Node{}
	tail := &Node{}
	head.next = tail
	tail.prev = head
	return MyLinkedList{head: head, tail: tail, size: 0}
}

func (this *MyLinkedList) getNode(index int) *Node {
	p := this.head.next
	// TODO: 可以优化，通过 index 判断从 head 还是 tail 开始遍历
	for range index {
		p = p.next
	}
	return p
}
func (this *MyLinkedList) isElementIndex(index int) bool {
	return index >= 0 && index < this.size
}

func (this *MyLinkedList) checkElementIndex(index int) error {
	if !this.isElementIndex(index) {
		return fmt.Errorf("Index: %d, Size: %d", index, this.size)
	}
	return nil
}

func (this *MyLinkedList) Get(index int) int {
	if err := this.checkElementIndex(index); err != nil {
		return -1
	}
	// 找到 index 对应的 Node
	p := this.getNode(index)

	return p.val

}

func (this *MyLinkedList) AddAtHead(val int) {
	x := &Node{val: val}
	temp := this.head.next
	// head <-> temp
	temp.prev = x
	x.next = temp

	this.head.next = x
	x.prev = this.head
	// head <-> x <-> temp
	this.size++

}

func (this *MyLinkedList) AddAtTail(val int) {
	x := &Node{val: val}
	preNode := this.tail.prev
	preNode.next = x
	x.prev = preNode
	this.tail.prev = x
	x.next = this.tail
	this.size++
}

func (this *MyLinkedList) AddAtIndex(index int, val int) {
	if index < 0 || index > this.size {
		return
	}
	if index == this.size {
		this.AddAtTail(val)
		return
	}

	// 找到 index 对应的 Node
	p := this.getNode(index)
	temp := p.prev
	// temp <-> p

	// 新要插入的 Node
	x := &Node{val: val}

	p.prev = x
	temp.next = x

	x.prev = temp
	x.next = p

	// temp <-> x <-> p

	this.size++

}

func (this *MyLinkedList) DeleteAtIndex(index int) {
	if err := this.checkElementIndex(index); err != nil {
		return
	}
	p := this.getNode(index)
	// prev <-> p <-> next
	p.prev.next = p.next
	p.next.prev = p.prev
	// prev <-> next
	this.size--
}

```
