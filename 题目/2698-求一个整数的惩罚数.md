# 2698. 求一个整数的惩罚数

[LeetCode 链接](https://leetcode.cn/problems/find-the-punishment-number-of-an-integer/)

## 来源：灵茶山艾府和代码随想录 / 回溯

组合分割问题，注意不要计算重复，并且还要注意，获取nums的时候，获得的数字是反过来的。
```c++
class Solution {  
public:  
    void getNums(vector<int>& nums, int n) {  
        while (n > 0) {  
            nums.push_back(n % 10);  
            n /= 10;  
        }  
        reverse(nums.begin(), nums.end());  
    }  
  
    int getSum(vector<int>& nums) {  
        int sum = 0;  
        for (int i = nums.size() - 1; i >= 0; i--) {  
            sum += nums[i];  
        }  
        return sum;  
    }  
  
    int getNum(vector<int>& nums, int left, int right) {  
        int num = 0;  
        for (int i = left; i <= right; ++i) {  
            num = num * 10 + nums[i];  // 改进这里的计算方法  
        }  
        return num;  
    }  
  
    bool bT9(vector<int>& nums, int num, vector<int>& path, int i, int& res) {  
        if (i == nums.size()) {  
            if (getSum(path) == num) {  
                res += num * num;  
                return true;  
            }  
            return false;  
        }  
        for (int j = i; j < nums.size(); j++) {  
            path.push_back(getNum(nums, i, j));  
            if (bT9(nums, num, path, j + 1, res)) return true;  
            path.pop_back();  
        }  
        return false;  
    }  
  
    int punishmentNumber(int n) {  
        int ans = 0;  
        for (int i = 1; i <= n; i++) {  // Changed from i < n to i <= n  
            vector<int> nums;  
            getNums(nums, i * i);  
            vector<int> path;  
            bT9(nums, i, path, 0, ans);  
        }  
        return ans;  
    }  
};
```
